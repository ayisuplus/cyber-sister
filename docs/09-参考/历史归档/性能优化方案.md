## 二、缓存优化

### 2.1 多级缓存架构

```javascript
// 多级缓存管理器
class MultiLevelCache {
  constructor() {
    // L1: 本地缓存 (进程内)
    this.l1Cache = new LRUCache({
      max: 1000,
      ttl: 1000 * 60 * 5 // 5分钟
    });
    
    // L2: Redis缓存 (分布式)
    this.l2Cache = new Redis();
    
    // L3: 数据库
    this.l3Cache = prisma;
  }

  // 获取数据
  async get(key) {
    // L1: 本地缓存
    const l1Result = this.l1Cache.get(key);
    if (l1Result) {
      return l1Result;
    }

    // L2: Redis缓存
    const l2Result = await this.l2Cache.get(key);
    if (l2Result) {
      const data = JSON.parse(l2Result);
      this.l1Cache.set(key, data);
      return data;
    }

    // L3: 数据库
    return null;
  }

  // 设置数据
  async set(key, value, ttl = 3600) {
    // L1: 本地缓存
    this.l1Cache.set(key, value);
    
    // L2: Redis缓存
    await this.l2Cache.setex(key, ttl, JSON.stringify(value));
  }

  // 删除数据
  async delete(key) {
    this.l1Cache.delete(key);
    await this.l2Cache.del(key);
  }

  // 批量获取
  async mget(keys) {
    // L1: 本地缓存
    const l1Results = {};
    const missingKeys = [];
    
    for (const key of keys) {
      const l1Result = this.l1Cache.get(key);
      if (l1Result) {
        l1Results[key] = l1Result;
      } else {
        missingKeys.push(key);
      }
    }

    if (missingKeys.length === 0) {
      return l1Results;
    }

    // L2: Redis缓存
    const l2Results = await this.l2Cache.mget(missingKeys);
    const l2Data = {};
    
    for (let i = 0; i < missingKeys.length; i++) {
      if (l2Results[i]) {
        const data = JSON.parse(l2Results[i]);
        l2Data[missingKeys[i]] = data;
        this.l1Cache.set(missingKeys[i], data);
      }
    }

    return { ...l1Results, ...l2Data };
  }

  // 批量设置
  async mset(keyValuePairs, ttl = 3600) {
    const pipeline = this.l2Cache.pipeline();
    
    for (const [key, value] of Object.entries(keyValuePairs)) {
      this.l1Cache.set(key, value);
      pipeline.setex(key, ttl, JSON.stringify(value));
    }
    
    await pipeline.exec();
  }
}

// 使用示例
const cache = new MultiLevelCache();

// 获取用户数据
const user = await cache.get(`user:${userId}`);
if (!user) {
  const user = await prisma.user.findUnique({ where: { id: userId } });
  await cache.set(`user:${userId}`, user, 3600);
}
```

### 2.2 缓存策略

```javascript
// 缓存策略管理器
class CacheStrategy {
  constructor() {
    this.cache = new MultiLevelCache();
  }

  // Cache-Aside 模式
  async cacheAside(key, fetchFn, ttl = 3600) {
    // 1. 先查缓存
    let data = await this.cache.get(key);
    
    if (data) {
      return data;
    }

    // 2. 缓存未命中，查数据库
    data = await fetchFn();
    
    if (data) {
      // 3. 写入缓存
      await this.cache.set(key, data, ttl);
    }

    return data;
  }

  // Write-Through 模式
  async writeThrough(key, data, writeFn, ttl = 3600) {
    // 1. 写入数据库
    await writeFn(data);
    
    // 2. 写入缓存
    await this.cache.set(key, data, ttl);
    
    return data;
  }

  // Write-Behind 模式
  async writeBehind(key, data, writeFn, ttl = 3600) {
    // 1. 写入缓存
    await this.cache.set(key, data, ttl);
    
    // 2. 异步写入数据库
    setImmediate(async () => {
      try {
        await writeFn(data);
      } catch (error) {
        console.error('Write-behind failed:', error);
        // 可以添加重试逻辑
      }
    });
    
    return data;
  }

  // Read-Through 模式
  async readThrough(key, fetchFn, ttl = 3600) {
    // 1. 先查缓存
    let data = await this.cache.get(key);
    
    if (data) {
      return data;
    }

    // 2. 缓存未命中，查数据库并写入缓存
    data = await fetchFn();
    
    if (data) {
      await this.cache.set(key, data, ttl);
    }

    return data;
  }

  // 缓存预热
  async warmup(keys, fetchFn) {
    const batchSize = 100;
    
    for (let i = 0; i < keys.length; i += batchSize) {
      const batch = keys.slice(i, i + batchSize);
      const data = await fetchFn(batch);
      
      await this.cache.mset(data);
    }
  }

  // 缓存降级
  async degrade(key, fallbackFn) {
    try {
      const data = await this.cache.get(key);
      if (data) {
        return data;
      }
    } catch (error) {
      console.error('Cache error:', error);
    }

    // 降级到数据库
    return await fallbackFn();
  }
}

// 使用示例
const cacheStrategy = new CacheStrategy();

// Cache-Aside 模式
const user = await cacheStrategy.cacheAside(
  `user:${userId}`,
  () => prisma.user.findUnique({ where: { id: userId } }),
  3600
);

// Write-Through 模式
await cacheStrategy.writeThrough(
  `user:${userId}`,
  userData,
  (data) => prisma.user.update({ where: { id: userId }, data }),
  3600
);
```

### 3.1 查询优化

```javascript
// 查询优化器
class QueryOptimizer {
  constructor() {
    this.prisma = new PrismaClient();
  }

  // 分页查询优化
  async paginate(model, options = {}) {
    const { page = 1, pageSize = 20, where = {}, orderBy = {} } = options;
    
    // 使用游标分页（更高效）
    if (options.cursor) {
      const results = await this.prisma[model].findMany({
        where,
        orderBy,
        take: pageSize + 1,
        cursor: { id: options.cursor },
        skip: 1
      });
      
      const hasNextPage = results.length > pageSize;
      const data = hasNextPage ? results.slice(0, -1) : results;
      
      return {
        data,
        hasNextPage,
        nextCursor: hasNextPage ? data[data.length - 1].id : null
      };
    }
    
    // 传统分页
    const skip = (page - 1) * pageSize;
    
    const [data, total] = await Promise.all([
      this.prisma[model].findMany({
        where,
        orderBy,
        skip,
        take: pageSize
      }),
      this.prisma[model].count({ where })
    ]);
    
    return {
      data,
      total,
      page,
      pageSize,
      totalPages: Math.ceil(total / pageSize)
    };
  }

  // 批量查询优化
  async batchQuery(queries) {
    return await this.prisma.$transaction(
      queries.map(query => {
        const { model, method, args } = query;
        return this.prisma[model][method](args);
      })
    );
  }

  // 查询重写优化
  async optimizedFind(model, options = {}) {
    const { select, include, where, orderBy, take, skip } = options;
    
    // 只选择需要的字段
    const optimizedSelect = select || this.getOptimalSelect(model, options);
    
    // 优化include
    const optimizedInclude = include ? this.optimizeInclude(include) : undefined;
    
    return await this.prisma[model].findMany({
      select: optimizedSelect,
      include: optimizedInclude,
      where,
      orderBy,
      take,
      skip
    });
  }

  // 获取最优字段选择
  getOptimalSelect(model, options) {
    // 根据查询场景选择字段
    const baseFields = {
      user: ['id', 'phone', 'nickname', 'persona', 'is_vip', 'created_at'],
      conversation: ['id', 'title', 'persona', 'created_at'],
      message: ['id', 'role', 'content', 'emotion', 'importance', 'created_at'],
      memory: ['id', 'type', 'content', 'importance', 'created_at']
    };
    
    return baseFields[model] || undefined;
  }

  // 优化include
  optimizeInclude(include) {
    const optimized = {};
    
    for (const [key, value] of Object.entries(include)) {
      if (value === true) {
        optimized[key] = { select: { id: true } };
      } else if (typeof value === 'object') {
        optimized[key] = this.optimizeInclude(value);
      }
    }
    
    return optimized;
  }
}

// 使用示例
const queryOptimizer = new QueryOptimizer();

// 分页查询
const users = await queryOptimizer.paginate('user', {
  page: 1,
  pageSize: 20,
  where: { is_vip: true },
  orderBy: { created_at: 'desc' }
});

// 批量查询
const results = await queryOptimizer.batchQuery([
  { model: 'user', method: 'findUnique', args: { where: { id: userId } } },
  { model: 'conversation', method: 'findMany', args: { where: { userId } } }
]);
```

// 数据库连接池管理
class DatabasePool {
  constructor() {
    this.prisma = new PrismaClient({
      datasources: {
        db: {
          url: process.env.DATABASE_URL
        }
      },
      log: ['query', 'info', 'warn', 'error']
    });
    
    // 连接池配置
    this.poolConfig = {
      connection_limit: 20,
      pool_timeout: 30,
      connect_timeout: 10
    };
  }

  // 查询性能监控
  async queryWithMetrics(queryFn, queryName) {
    const start = Date.now();
    
    try {
      const result = await queryFn(this.prisma);
      const duration = Date.now() - start;
      
      // 记录查询性能
      this.recordQueryMetrics(queryName, duration, 'success');
      
      // 慢查询告警
      if (duration > 1000) {
        console.warn(`Slow query detected: ${queryName} took ${duration}ms`);
      }
      
      return result;
    } catch (error) {
      const duration = Date.now() - start;
      this.recordQueryMetrics(queryName, duration, 'error');
      throw error;
    }
  }

  // 记录查询指标
  recordQueryMetrics(queryName, duration, status) {
    // 发送到Prometheus
    prometheus.queryDuration.observe({ query: queryName }, duration / 1000);
    prometheus.queryCount.inc({ query: queryName, status });
  }

  // 连接池健康检查
  async healthCheck() {
    try {
      await this.prisma.$queryRaw`SELECT 1`;
      return { status: 'healthy', timestamp: new Date() };
    } catch (error) {
      return { status: 'unhealthy', error: error.message, timestamp: new Date() };
    }
  }

  // 连接池统计
  async getPoolStats() {
    const stats = await this.prisma.$metrics.json();
    return {
      activeConnections: stats.gauges.find(m => m.name === 'prisma_pool_connections_open')?.value || 0,
      idleConnections: stats.gauges.find(m => m.name === 'prisma_pool_connections_idle')?.value || 0,
      waitingRequests: stats.gauges.find(m => m.name === 'prisma_pool_connections_busy')?.value || 0
    };
  }
}

// 使用示例

  (prisma) => prisma.user.findUnique({ where: { id: userId } }),
  'findUserById'
);

// 异步任务队列
class AsyncTaskQueue {
  constructor() {
    this.queue = [];
    this.processing = false;
    this.concurrency = 5;
    this.activeTasks = 0;
  }

  // 添加任务
  async add(task, priority = 'normal') {
    return new Promise((resolve, reject) => {
      this.queue.push({
        task,
        priority,
        resolve,
        reject,
        createdAt: Date.now()
      });
      
      // 按优先级排序
      this.queue.sort((a, b) => {
        const priorityOrder = { high: 0, normal: 1, low: 2 };
        return priorityOrder[a.priority] - priorityOrder[b.priority];
      });
      
      this.processQueue();
    });
  }

  // 处理队列
  async processQueue() {
    if (this.processing || this.activeTasks >= this.concurrency) {
      return;
    }
    
    this.processing = true;
    
    while (this.queue.length > 0 && this.activeTasks < this.concurrency) {
      const item = this.queue.shift();
      this.activeTasks++;
      
      this.processTask(item).finally(() => {
        this.activeTasks--;
        this.processQueue();
      });
    }
    
    this.processing = false;
  }

  // 处理单个任务
  async processTask(item) {
    try {
      const result = await item.task();
      item.resolve(result);
    } catch (error) {
      item.reject(error);
    }
  }

  // 获取队列状态
  getStatus() {
    return {
      queueLength: this.queue.length,
      activeTasks: this.activeTasks,
      concurrency: this.concurrency
    };
  }
}

// 使用示例
const taskQueue = new AsyncTaskQueue();

// 添加异步任务
const result = await taskQueue.add(
  () => heavyComputation(),
  'high'
);

// 消息队列消费者
class MessageQueueConsumer {
  constructor() {
    this.redis = new Redis();
    this.handlers = new Map();
    this.consumers = new Map();
  }

  // 注册处理器
  registerHandler(stream, handler) {
    this.handlers.set(stream, handler);
  }

  // 启动消费者
  async startConsumer(stream, group, consumer, concurrency = 5) {
    const handler = this.handlers.get(stream);
    if (!handler) {
      throw new Error(`No handler registered for stream: ${stream}`);
    }

    // 创建消费者组
    try {
      await this.redis.xgroup('CREATE', stream, group, '0', 'MKSTREAM');
    } catch (error) {
      if (!error.message.includes('BUSYGROUP')) {
        throw error;
      }
    }

    // 启动多个消费者
    for (let i = 0; i < concurrency; i++) {
      const consumerName = `${consumer}-${i}`;
      this.consumeMessages(stream, group, consumerName, handler);
    }
  }

  // 消费消息
  async consumeMessages(stream, group, consumer, handler) {
    while (true) {
      try {
        const messages = await this.redis.xreadgroup(
          'GROUP', group, consumer,
          'COUNT', 10,
          'BLOCK', 2000,
          'STREAMS', stream, '>'
        );

        if (messages) {
          for (const [streamName, entries] of messages) {
            for (const [id, fields] of entries) {
              try {
                const data = JSON.parse(fields.data);
                await handler(data, id);
                await this.redis.xack(streamName, group, id);
              } catch (error) {
                console.error('Message processing error:', error);
                // 将失败消息放入死信队列
                await this.redis.xadd(`${stream}:dead`, '*', 'data', fields.data);
              }
            }
          }
        }
      } catch (error) {
        console.error('Consumer error:', error);
        await new Promise(resolve => setTimeout(resolve, 1000));
      }
    }
  }

  // 停止消费者
  async stopConsumer(stream, group, consumer) {
    // 实现停止逻辑
  }
}

// 使用示例
const consumer = new MessageQueueConsumer();

// 注册处理器
consumer.registerHandler('chat-messages', async (data, id) => {
  // 处理聊天消息
  await processChatMessage(data);
});

// 启动消费者
await consumer.startConsumer('chat-messages', 'memory-group', 'memory-consumer', 5);

// LLM服务优化
class LLMServiceOptimizer {
  constructor() {
    this.circuitBreaker = new CircuitBreaker({
      failureThreshold: 5,
      resetTimeout: 60000
    });
    
    this.rateLimiter = new RateLimiter({
      windowMs: 60000,
      max: 100
    });
    
    this.cache = new Redis();
  }

  // 优化后的推理调用
  async generate(message, options = {}) {
    // 1. 检查缓存
    const cacheKey = this.getCacheKey(message, options);
    const cached = await this.cache.get(cacheKey);
    if (cached) {
      return cached;
    }

    // 2. 限流检查
    await this.rateLimiter.check(options.userId);

    // 3. 熔断器检查
    return await this.circuitBreaker.execute(async () => {
      // 4. 调用LLM服务
      const response = await axios.post('http://llm-service/api/generate', {
        message,
        ...options
      }, {
        timeout: 30000
      });

      // 5. 缓存结果
      await this.cache.setex(cacheKey, 3600, response.data);

      return response.data;
    });
  }

  // 生成缓存键
  getCacheKey(message, options) {
    const key = `${message}:${options.persona}:${options.emotion}`;
    return `llm:${crypto.createHash('md5').update(key).digest('hex')}`;
  }

      })

  recordDatabaseQuery(query, duration) {
    this.metrics.databaseQueryDuration.observe({ query }, duration);

  }

    this.metrics.activeConnections.set(count);
  }

      res.on('finish', () => {
        const duration = (Date.now() - start) / 1000;
        this.recordRequest(req.method, req.path, res.statusCode, duration);
      });
      
      next();
    };
  }
}

// 性能分析工具
class PerformanceAnalyzer {
  constructor() {
    this.profiles = new Map();
  }

  // 开始分析
  startProfile(name) {
    this.profiles.set(name, {
      start: Date.now(),
      marks: []
    });
  }

  // 标记时间点
  mark(name, label) {
    const profile = this.profiles.get(name);
    if (profile) {
      profile.marks.push({
        label,
        timestamp: Date.now(),
        duration: Date.now() - profile.start
      });
    }
  }

  // 结束分析
  endProfile(name) {
    const profile = this.profiles.get(name);
    if (profile) {
      const duration = Date.now() - profile.start;
      this.profiles.delete(name);
      
      return {
        name,
        duration,
        marks: profile.marks
      };
    }
    return null;
  }

  // 分析函数执行时间
  async analyze(name, fn) {
    this.startProfile(name);
    
    try {
      const result = await fn();
      this.mark(name, 'execution');
      return result;
    } finally {
      const profile = this.endProfile(name);
      console.log(`Performance Profile [${name}]:`, profile);
    }
  }
}

// 使用示例

  analyzer.mark('getUserProfile', 'dbQuery');
  
  const memories = await memoryService.getUserMemories(userId);
  analyzer.mark('getUserProfile', 'memoryQuery');
  
  return { user, memories };
});
