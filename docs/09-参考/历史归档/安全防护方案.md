### 2.1 JWT认证

```javascript
// JWT认证服务
const jwt = require('jsonwebtoken');
const crypto = require('crypto');

class JWTService {
  constructor() {
    this.secret = process.env.JWT_SECRET;
    this.refreshSecret = process.env.JWT_REFRESH_SECRET;
    this.accessTokenExpiry = '15m';
    this.refreshTokenExpiry = '7d';
  }

  // 生成访问令牌
  generateAccessToken(payload) {
    return jwt.sign(payload, this.secret, {
      expiresIn: this.accessTokenExpiry,
      issuer: 'cyber-sister',
      audience: 'cyber-sister-api'
    });
  }

  // 生成刷新令牌
  generateRefreshToken(payload) {
    return jwt.sign(payload, this.refreshSecret, {
      expiresIn: this.refreshTokenExpiry,
      issuer: 'cyber-sister',
      audience: 'cyber-sister-api'
    });
  }

  // 生成令牌对
  generateTokenPair(user) {
    const payload = {
      userId: user.id,
      phone: user.phone,
      persona: user.persona,
      isVip: user.isVip
    };

    return {
      accessToken: this.generateAccessToken(payload),
      refreshToken: this.generateRefreshToken(payload),
      expiresIn: 900 // 15分钟
    };
  }

  // 验证访问令牌
  verifyAccessToken(token) {
    try {
      return jwt.verify(token, this.secret, {
        issuer: 'cyber-sister',
        audience: 'cyber-sister-api'
      });
    } catch (error) {
      throw new Error('Invalid access token');
    }
  }

  // 验证刷新令牌
  verifyRefreshToken(token) {
    try {
      return jwt.verify(token, this.refreshSecret, {
        issuer: 'cyber-sister',
        audience: 'cyber-sister-api'
      });
    } catch (error) {
      throw new Error('Invalid refresh token');
    }
  }

  // 刷新令牌
  async refreshToken(refreshToken) {
    const decoded = this.verifyRefreshToken(refreshToken);
    
    // 检查刷新令牌是否在黑名单中
    const isBlacklisted = await this.isTokenBlacklisted(refreshToken);
    if (isBlacklisted) {
      throw new Error('Refresh token has been revoked');
    }

    // 生成新的令牌对
    const user = await prisma.user.findUnique({
      where: { id: decoded.userId }
    });

    if (!user) {
      throw new Error('User not found');
    }

    // 将旧的刷新令牌加入黑名单
    await this.blacklistToken(refreshToken);

    return this.generateTokenPair(user);
  }

  // 将令牌加入黑名单
  async blacklistToken(token) {
    const decoded = jwt.decode(token);
    const expiry = decoded.exp - Math.floor(Date.now() / 1000);
    
    if (expiry > 0) {
      await redis.setex(`blacklist:${token}`, expiry, '1');
    }
  }

  // 检查令牌是否在黑名单中
  async isTokenBlacklisted(token) {
    const result = await redis.get(`blacklist:${token}`);
    return result === '1';
  }
}

// 认证中间件
function authenticate(req, res, next) {
  const authHeader = req.headers.authorization;
  
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({
      error: 'Unauthorized',
      code: 'MISSING_TOKEN'
    });
  }

  const token = authHeader.substring(7);
  
  try {
    const decoded = jwtService.verifyAccessToken(token);
    req.user = decoded;
    next();
  } catch (error) {
    return res.status(401).json({
      error: 'Unauthorized',
      code: 'INVALID_TOKEN'
    });
  }
}

// 授权中间件
function authorize(...roles) {
  return (req, res, next) => {
    if (!req.user) {
      return res.status(401).json({
        error: 'Unauthorized',
        code: 'NOT_AUTHENTICATED'
      });
    }

    if (roles.length > 0 && !roles.includes(req.user.role)) {
      return res.status(403).json({
        error: 'Forbidden',
        code: 'INSUFFICIENT_PERMISSIONS'
      });
    }

    next();
  };
}

// 使用示例
const jwtService = new JWTService();

// 登录
app.post('/api/auth/login', async (req, res) => {
  const { phone, code } = req.body;
  
  // 验证验证码
  const isValid = await verifyCode(phone, code);
  if (!isValid) {
    return res.status(400).json({ error: 'Invalid verification code' });
  }

  // 查找或创建用户
  let user = await prisma.user.findUnique({ where: { phone } });
  if (!user) {
    user = await prisma.user.create({ data: { phone } });
  }

  // 生成令牌
  const tokens = jwtService.generateTokenPair(user);
  
  res.json(tokens);
});

// 受保护的路由
app.get('/api/user/profile', authenticate, async (req, res) => {
  const user = await prisma.user.findUnique({
    where: { id: req.user.userId }
  });
  res.json(user);
});

// 需要特定角色的路由
app.delete('/api/admin/users/:id', authenticate, authorize('admin'), async (req, res) => {
  await prisma.user.delete({ where: { id: req.params.id } });
  res.json({ success: true });
});
```

### 2.2 验证码服务

```javascript
// 验证码服务
class VerificationCodeService {
  constructor() {
    this.redis = new Redis();
    this.codeLength = 6;
    this.codeExpiry = 300; // 5分钟
    this.maxAttempts = 5;
    this.cooldownPeriod = 60; // 1分钟
  }

  // 生成验证码
  generateCode() {
    return crypto.randomInt(100000, 999999).toString();
  }

  // 发送验证码
  async sendCode(phone) {
    // 检查冷却期
    const cooldownKey = `cooldown:${phone}`;
    const cooldown = await this.redis.get(cooldownKey);
    if (cooldown) {
      throw new Error('Please wait before requesting another code');
    }

    // 生成验证码
    const code = this.generateCode();
    
    // 存储验证码
    const codeKey = `code:${phone}`;
    await this.redis.setex(codeKey, this.codeExpiry, JSON.stringify({
      code,
      attempts: 0,
      createdAt: Date.now()
    }));

    // 设置冷却期
    await this.redis.setex(cooldownKey, this.cooldownPeriod, '1');

    // 发送短信
    await this.sendSMS(phone, code);

    return { success: true, message: 'Verification code sent' };
  }

  // 验证验证码
  async verifyCode(phone, code) {
    const codeKey = `code:${phone}`;
    const stored = await this.redis.get(codeKey);
    
    if (!stored) {
      throw new Error('Verification code expired or not found');
    }

    const data = JSON.parse(stored);
    
    // 检查尝试次数
    if (data.attempts >= this.maxAttempts) {
      await this.redis.del(codeKey);
      throw new Error('Too many attempts, please request a new code');
    }

    // 增加尝试次数
    data.attempts++;
    await this.redis.setex(codeKey, this.codeExpiry, JSON.stringify(data));

    // 验证验证码
    if (data.code !== code) {
      throw new Error('Invalid verification code');
    }

    // 验证成功，删除验证码
    await this.redis.del(codeKey);

    return true;
  }

  // 发送短信
  async sendSMS(phone, code) {
    // 调用短信服务API
    const response = await axios.post('https://sms-api.example.com/send', {
      phone,
      message: `您的验证码是：${code}，5分钟内有效。`
    });

    if (response.data.code !== 200) {
      throw new Error('Failed to send SMS');
    }
  }
}

// 使用示例
const verificationCodeService = new VerificationCodeService();

// 发送验证码
app.post('/api/auth/send-code', async (req, res) => {
  try {
    const { phone } = req.body;
    await verificationCodeService.sendCode(phone);
    res.json({ success: true, message: 'Verification code sent' });
  } catch (error) {
    res.status(400).json({ error: error.message });
  }
});

// 验证验证码
app.post('/api/auth/verify-code', async (req, res) => {
  try {
    const { phone, code } = req.body;
    await verificationCodeService.verifyCode(phone, code);
    res.json({ success: true, message: 'Verification code verified' });
  } catch (error) {
    res.status(400).json({ error: error.message });
  }
});
```

```javascript
// 数据加密服务
const crypto = require('crypto');

class EncryptionService {
  constructor() {
    this.algorithm = 'aes-256-gcm';
    this.secretKey = Buffer.from(process.env.ENCRYPTION_KEY, 'hex');
    this.ivLength = 16;
    this.tagLength = 16;
  }

  // 加密数据
  encrypt(text) {
    const iv = crypto.randomBytes(this.ivLength);
    const cipher = crypto.createCipheriv(this.algorithm, this.secretKey, iv);
    
    let encrypted = cipher.update(text, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    
    const tag = cipher.getAuthTag();
    
    return {
      iv: iv.toString('hex'),
      encrypted,
      tag: tag.toString('hex')
    };
  }

  // 解密数据
  decrypt(encryptedData) {
    const iv = Buffer.from(encryptedData.iv, 'hex');
    const tag = Buffer.from(encryptedData.tag, 'hex');
    const decipher = crypto.createDecipheriv(this.algorithm, this.secretKey, iv);
    
    decipher.setAuthTag(tag);
    
    let decrypted = decipher.update(encryptedData.encrypted, 'hex', 'utf8');
    decrypted += decipher.final('utf8');
    
    return decrypted;
  }

  // 哈希数据
  hash(data) {
    return crypto.createHash('sha256').update(data).digest('hex');
  }

  // 生成随机密钥
  generateKey() {
    return crypto.randomBytes(32).toString('hex');
  }

  // 密码哈希
  async hashPassword(password) {
    const bcrypt = require('bcryptjs');
    const salt = await bcrypt.genSalt(12);
    return await bcrypt.hash(password, salt);
  }

  // 验证密码
  async verifyPassword(password, hash) {
    const bcrypt = require('bcryptjs');
    return await bcrypt.compare(password, hash);
  }
}

// 使用示例
const encryptionService = new EncryptionService();

// 加密敏感数据
const encryptedPhone = encryptionService.encrypt('13800138000');
console.log('Encrypted:', encryptedPhone);

// 解密数据
const decryptedPhone = encryptionService.decrypt(encryptedPhone);
console.log('Decrypted:', decryptedPhone);

// 哈希数据
const hash = encryptionService.hash('sensitive-data');
console.log('Hash:', hash);
```

```javascript
// 数据脱敏服务
class DataMaskingService {
  constructor() {
    this.patterns = {
      phone: /(\d{3})\d{4}(\d{4})/,
      email: /(.{2})(.*)(@.*)/,
      idCard: /(\d{4})\d{10}(\d{4})/,
      bankCard: /(\d{4})\d+(\d{4})/
    };
  }

  // 手机号脱敏
  maskPhone(phone) {
    return phone.replace(this.patterns.phone, '$1****$2');
  }

  // 邮箱脱敏
  maskEmail(email) {
    return email.replace(this.patterns.email, '$1***$3');
  }

  // 身份证脱敏
  maskIdCard(idCard) {
    return idCard.replace(this.patterns.idCard, '$1**********$2');
  }

  // 银行卡脱敏
  maskBankCard(bankCard) {
    return bankCard.replace(this.patterns.bankCard, '$1 **** **** $2');
  }

  // 姓名脱敏
  maskName(name) {
    if (name.length <= 2) {
      return name.charAt(0) + '*';
    }
    return name.charAt(0) + '*'.repeat(name.length - 2) + name.charAt(name.length - 1);
  }

  // 通用脱敏
  mask(data, type) {
    switch (type) {
      case 'phone':
        return this.maskPhone(data);
      case 'email':
        return this.maskEmail(data);
      case 'idCard':
        return this.maskIdCard(data);
      case 'bankCard':
        return this.maskBankCard(data);
      case 'name':
        return this.maskName(data);
      default:
        return data;
    }
  }

  // 批量脱敏
  maskBatch(data, rules) {
    const masked = { ...data };
    
    for (const [field, type] of Object.entries(rules)) {
      if (masked[field]) {
        masked[field] = this.mask(masked[field], type);
      }
    }
    
    return masked;
  }
}

// 使用示例
const maskingService = new DataMaskingService();

// 手机号脱敏
const maskedPhone = maskingService.maskPhone('13800138000');
console.log('Masked phone:', maskedPhone); // 138****8000

// 批量脱敏
const user = {
  name: '张三',
  phone: '13800138000',
  email: 'zhangsan@example.com'
};

const maskedUser = maskingService.maskBatch(user, {
  name: 'name',
  phone: 'phone',
  email: 'email'
});

console.log('Masked user:', maskedUser);
```

```javascript
// 输入验证服务
const Joi = require('joi');

class InputValidationService {
  constructor() {
    this.schemas = this.defineSchemas();
  }

  // 定义验证模式
  defineSchemas() {
    return {
      // 用户注册
      register: Joi.object({
        phone: Joi.string().pattern(/^1[3-9]\d{9}$/).required(),
        code: Joi.string().length(6).required()
      }),

      // 用户登录
      login: Joi.object({
        phone: Joi.string().pattern(/^1[3-9]\d{9}$/).required(),
        code: Joi.string().length(6).required()
      }),

      // 发送验证码
      sendCode: Joi.object({
        phone: Joi.string().pattern(/^1[3-9]\d{9}$/).required()
      }),

      // 更新用户信息
      updateProfile: Joi.object({
        nickname: Joi.string().min(2).max(50).optional(),
        persona: Joi.string().valid('toxic', 'gentle', 'wild').optional(),
        avatar_url: Joi.string().uri().optional()
      }),

      // 发送消息
      sendMessage: Joi.object({
        content: Joi.string().min(1).max(10000).required(),
        conversationId: Joi.string().uuid().required()
      }),

      // 创建待办
      createTodo: Joi.object({
        content: Joi.string().min(1).max(500).required(),
        due_date: Joi.date().iso().optional()
      }),

      // 创建倒数日
      createCountdown: Joi.object({
        title: Joi.string().min(1).max(100).required(),
        target_date: Joi.date().iso().required()
      })
    };
  }

  // 验证数据
  validate(schemaName, data) {
    const schema = this.schemas[schemaName];
    if (!schema) {
      throw new Error(`Schema ${schemaName} not found`);
    }

    const { error, value } = schema.validate(data, {
      abortEarly: false,
      stripUnknown: true
    });

    if (error) {
      const errors = error.details.map(detail => ({
        field: detail.path.join('.'),
        message: detail.message
      }));
      throw new ValidationError('Validation failed', errors);
    }

    return value;
  }

  // 验证中间件
  validateMiddleware(schemaName) {
    return (req, res, next) => {
      try {
        req.body = this.validate(schemaName, req.body);
        next();
      } catch (error) {
        if (error instanceof ValidationError) {
          return res.status(400).json({
            error: 'Validation failed',
            details: error.errors
          });
        }
        next(error);
      }
    };
  }
}

// 使用示例
const validationService = new InputValidationService();

// 注册验证

  validationService.validateMiddleware('register'),
  async (req, res) => {
    // 处理注册逻辑
  }
);

// 发送消息验证
app.post('/api/chat/messages',
  authenticate,
  validationService.validateMiddleware('sendMessage'),
  async (req, res) => {
    // 处理发送消息逻辑
  }
);
```

```javascript
// 限流服务
const rateLimit = require('express-rate-limit');
const RedisStore = require('rate-limit-redis');

class RateLimitService {
  constructor() {
    this.redis = new Redis();
  }

  // 创建限流器
  createLimiter(options = {}) {
    return rateLimit({
      store: new RedisStore({
        sendCommand: (...args) => this.redis.call(...args)
      }),
      windowMs: options.windowMs || 15 * 60 * 1000,
      max: options.max || 100,
      message: options.message || 'Too many requests',
      standardHeaders: true,
      legacyHeaders: false,
      keyGenerator: (req) => {
        return req.ip || req.headers['x-forwarded-for'] || req.connection.remoteAddress;
      }
    });
  }

  // API限流
  apiLimiter() {
    return this.createLimiter({
      windowMs: 15 * 60 * 1000,
      max: 100
    });
  }

  // 登录限流
  loginLimiter() {
    return this.createLimiter({
      windowMs: 15 * 60 * 1000,
      max: 5,
      message: 'Too many login attempts'
    });
  }

  // 验证码限流
  codeLimiter() {
    return this.createLimiter({
      windowMs: 60 * 1000,
      max: 1,
      message: 'Please wait before requesting another code'
    });
  }

  // LLM调用限流
  llmLimiter() {
    return this.createLimiter({
      windowMs: 60 * 1000,
      max: 10,
      keyGenerator: (req) => req.user?.userId || req.ip
    });
  }
}

// 使用示例
const rateLimitService = new RateLimitService();

// API限流
app.use('/api/', rateLimitService.apiLimiter());

// 登录限流
app.use('/api/auth/login', rateLimitService.loginLimiter());

// 验证码限流
```

```javascript
// 审计日志服务
class AuditLogService {
  constructor() {
    this.prisma = new PrismaClient();
    this.redis = new Redis();
  }

  // 记录审计日志
  async log(action, data) {
    const auditLog = {
      userId: data.userId,
      action,
      resource: data.resource,
      resourceId: data.resourceId,
      details: data.details,
      ip: data.ip,
      userAgent: data.userAgent,
      timestamp: new Date()
    };

    // 存储到数据库
    await this.prisma.auditLog.create({
      data: auditLog
    });

    // 存储到Redis（用于实时监控）
    await this.redis.lpush('audit:logs', JSON.stringify(auditLog));
    await this.redis.ltrim('audit:logs', 0, 999);

    // 发送到监控系统
    await this.sendToMonitoring(auditLog);
  }

  // 记录用户操作
  async logUserAction(userId, action, details = {}) {
    await this.log('USER_ACTION', {
      userId,
      action,
      details,
      ip: details.ip,
      userAgent: details.userAgent
    });
  }

  // 记录系统操作
  async logSystemAction(action, details = {}) {
    await this.log('SYSTEM_ACTION', {
      userId: 'system',
      action,
      details
    });
  }

  // 记录安全事件
  async logSecurityEvent(event, details = {}) {
    await this.log('SECURITY_EVENT', {
      userId: details.userId || 'unknown',
      action: event,
      details,
      ip: details.ip,
      userAgent: details.userAgent
    });
  }

  // 查询审计日志
  async queryLogs(filters = {}) {
    const where = {};
    
    if (filters.userId) where.userId = filters.userId;
    if (filters.action) where.action = filters.action;
    if (filters.startDate) where.timestamp = { gte: filters.startDate };
    if (filters.endDate) where.timestamp = { ...where.timestamp, lte: filters.endDate };

    return await this.prisma.auditLog.findMany({
      where,
      orderBy: { timestamp: 'desc' },
      take: filters.limit || 100,
      skip: filters.offset || 0
    });
  }

  // 发送到监控系统
  async sendToMonitoring(auditLog) {
    // 发送到Prometheus
    prometheus.auditLogCount.inc({
      action: auditLog.action,
      resource: auditLog.resource
    });

    // 发送到ELK
    logger.info('Audit Log', auditLog);
  }
}
```

```javascript
// 合规检查服务
class ComplianceService {
  constructor() {
    this.redis = new Redis();
  }

  // 检查使用时长
  async checkUsageDuration(userId) {
    const key = `usage:${userId}`;
    const usage = await this.redis.get(key);
    
    if (!usage) {
      return { exceeded: false, duration: 0 };
    }

    const data = JSON.parse(usage);
    const duration = Date.now() - data.startTime;
    const twoHours = 2 * 60 * 60 * 1000;

    return {
      exceeded: duration >= twoHours,
      duration,
      startTime: data.startTime
    };
  }

  // 记录使用时长
  async recordUsage(userId) {
    const key = `usage:${userId}`;
    const usage = await this.redis.get(key);
    
    if (!usage) {
      await this.redis.set(key, JSON.stringify({
        startTime: Date.now(),
        lastActive: Date.now()
      }));
    } else {
      const data = JSON.parse(usage);
      data.lastActive = Date.now();
      await this.redis.set(key, JSON.stringify(data));
    }
  }

  // 清除使用记录
  async clearUsage(userId) {
    await this.redis.del(`usage:${userId}`);
  }

  // 检查危机内容
  async checkCrisisContent(content) {
    const crisisKeywords = [
      '自杀', '自残', '轻生', '不想活', '结束生命',
      '割腕', '跳楼', '上吊', '服毒', '投河'
    ];

    const lowerContent = content.toLowerCase();
    
    for (const keyword of crisisKeywords) {
      if (lowerContent.includes(keyword)) {
        return {
          isCrisis: true,
          level: 'high',
          keyword
        };
      }
    }

    return { isCrisis: false };
  }

  // 检查未成年人
  async checkMinor(userId) {
    const user = await prisma.user.findUnique({
      where: { id: userId },
      select: { birth_date: true }
    });

    if (!user || !user.birth_date) {
      return { isMinor: false, unknown: true };
    }

    const age = this.calculateAge(user.birth_date);
    return {
      isMinor: age < 18,
      age
    };
  }

  // 计算年龄
  calculateAge(birthDate) {
    const today = new Date();
    const birth = new Date(birthDate);
    let age = today.getFullYear() - birth.getFullYear();
    const monthDiff = today.getMonth() - birth.getMonth();
    
    if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birth.getDate())) {
      age--;
    }
    
    return age;
  }
}

// 使用示例
const complianceService = new ComplianceService();

if (usageCheck.exceeded) {
  // 发送提醒
  await notificationService.sendUsageReminder(userId);
}

if (crisisCheck.isCrisis) {
  // 触发危机干预
  await complianceService.triggerCrisisIntervention(userId, crisisCheck);
}
```

```javascript
// 密钥管理服务
class SecretManager {
  constructor() {
    this.vault = require('node-vault')({
      endpoint: process.env.VAULT_ENDPOINT
    });
  }

  // 获取密钥
  async getSecret(path) {
    try {
      const result = await this.vault.read(path);
      return result.data;
    } catch (error) {
      console.error('Failed to get secret:', error);
      throw error;
    }
  }

  // 设置密钥
  async setSecret(path, data) {
    try {
      await this.vault.write(path, data);
    } catch (error) {
      console.error('Failed to set secret:', error);
      throw error;
    }
  }

  // 删除密钥
  async deleteSecret(path) {
    try {
      await this.vault.delete(path);
    } catch (error) {
      console.error('Failed to delete secret:', error);
      throw error;
    }
  }

  // 生成动态密钥
  async generateDynamicSecret(role) {
    try {
      const result = await this.vault.write(`database/creds/${role}`, {});
      return {
        username: result.data.username,
        password: result.data.password,
        leaseId: result.lease_id,
        leaseDuration: result.lease_duration
      };
    } catch (error) {
      console.error('Failed to generate dynamic secret:', error);
      throw error;
    }
  }

  // 轮换密钥
  async rotateSecret(path) {
    try {
      const current = await this.getSecret(path);
      const newSecret = this.generateNewSecret();
      await this.setSecret(path, { ...current, ...newSecret });
      return newSecret;
    } catch (error) {
      console.error('Failed to rotate secret:', error);
      throw error;
    }
  }

  // 生成新密钥
  generateNewSecret() {
    const crypto = require('crypto');
    return {
      key: crypto.randomBytes(32).toString('hex'),
      rotatedAt: new Date().toISOString()
    };
  }
}

// 使用示例
const secretManager = new SecretManager();

// 获取数据库密码
const dbSecret = await secretManager.getSecret('secret/database');
const dbPassword = dbSecret.password;

// 生成动态数据库凭证
const dynamicCreds = await secretManager.generateDynamicSecret('readonly');
```
