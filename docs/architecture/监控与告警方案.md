# Amie监控与告警方案

> ⚠️ 参考：面向未来规模（如 DAU 5 万）的方案储备，非当前基线

**文档版本**：V1.0  
**设计日期**：2026年7月13日  
**设计目标**：全面监控、实时告警、快速定位、可视化展示

## 一、监控架构

### 1.1 监控体系概览

```
┌─────────────────────────────────────────────────────────────────────────────────┐
│                              监控体系                                           │
│  ┌──────────────────────────────────────────────────────────────────────────┐   │
│  │                         数据采集层                                        │   │
│  │    (Prometheus Exporter、Node Exporter、应用指标)                         │   │
│  └──────────────────────────────────────────────────────────────────────────┘   │
│  ┌──────────────────────────────────────────────────────────────────────────┐   │
│  │                         数据存储层                                        │   │
│  │              (Prometheus TSDB、Elasticsearch)                            │   │
│  └──────────────────────────────────────────────────────────────────────────┘   │
│  ┌──────────────────────────────────────────────────────────────────────────┐   │
│  │                         数据处理层                                        │   │
│  │           (Alertmanager告警、Logstash日志处理)                           │   │
│  └──────────────────────────────────────────────────────────────────────────┘   │
│  ┌──────────────────────────────────────────────────────────────────────────┐   │
│  │                         展示层                                            │   │
│  │           (Grafana仪表板、Kibana日志分析)                                │   │
│  └──────────────────────────────────────────────────────────────────────────┘   │
└─────────────────────────────────────────────────────────────────────────────────┘
```

### 1.2 监控指标分类

| 指标类别 | 指标名称 | 采集方式 | 告警阈值 |
|----------|----------|----------|----------|
| **系统指标** | CPU使用率 | Node Exporter | >80% |
| | 内存使用率 | Node Exporter | >85% |
| | 磁盘使用率 | Node Exporter | >90% |
| | 网络IO | Node Exporter | >100MB/s |
| **应用指标** | 请求速率 | Prometheus | >1000 QPS |
| | 错误率 | Prometheus | >5% |
| | 响应时间 | Prometheus | P95>200ms |
| | 活跃连接数 | Prometheus | >1000 |
| **数据库指标** | 连接数 | PostgreSQL Exporter | >150 |
| | 查询延迟 | PostgreSQL Exporter | >100ms |
| | 复制延迟 | PostgreSQL Exporter | >30s |
| | 缓存命中率 | Redis Exporter | <80% |
| **业务指标** | 活跃用户数 | 应用指标 | - |
| | 消息发送量 | 应用指标 | - |
| | 危机事件数 | 应用指标 | >0 |
| | LLM调用次数 | 应用指标 | - |

### 2.2 应用指标采集

```javascript
// 应用指标中间件
const prometheus = require('prom-client');

// 创建指标注册表
const register = new prometheus.Registry();

// 默认指标
prometheus.collectDefaultMetrics({ register });

// 自定义指标
const metrics = {
  // HTTP请求计数器
  httpRequestsTotal: new prometheus.Counter({
    name: 'http_requests_total',
    help: 'Total number of HTTP requests',
    labelNames: ['method', 'path', 'status'],
    registers: [register]
  }),

  // HTTP请求延迟直方图
  httpRequestDuration: new prometheus.Histogram({
    name: 'http_request_duration_seconds',
    help: 'HTTP request duration in seconds',
    labelNames: ['method', 'path'],
    buckets: [0.01, 0.05, 0.1, 0.5, 1, 2, 5],
    registers: [register]
  }),

  // 活跃连接数
  activeConnections: new prometheus.Gauge({
    name: 'active_connections',
    help: 'Number of active connections',
    registers: [register]
  }),

  // 数据库查询延迟
  databaseQueryDuration: new prometheus.Histogram({
    name: 'database_query_duration_seconds',
    help: 'Database query duration in seconds',
    labelNames: ['query', 'operation'],
    buckets: [0.001, 0.005, 0.01, 0.05, 0.1, 0.5, 1],
    registers: [register]
  }),

  // 缓存命中率
  cacheHitRate: new prometheus.Gauge({
    name: 'cache_hit_rate',
    help: 'Cache hit rate',
    labelNames: ['cache'],
    registers: [register]
  }),

  // LLM调用计数器
  llmCallsTotal: new prometheus.Counter({
    name: 'llm_calls_total',
    help: 'Total number of LLM calls',
    labelNames: ['model', 'status'],
    registers: [register]
  }),

  // LLM调用延迟
  llmCallDuration: new prometheus.Histogram({
    name: 'llm_call_duration_seconds',
    help: 'LLM call duration in seconds',
    labelNames: ['model'],
    buckets: [0.1, 0.5, 1, 2, 5, 10, 30],
    registers: [register]
  }),

  // 危机事件计数器
  crisisEventsTotal: new prometheus.Counter({
    name: 'crisis_events_total',
    help: 'Total number of crisis events',
    labelNames: ['level'],
    registers: [register]
  }),

  // 活跃用户数
  activeUsers: new prometheus.Gauge({
    name: 'active_users',
    help: 'Number of active users',
    registers: [register]
  }),

  // 消息发送量
  messagesSentTotal: new prometheus.Counter({
    name: 'messages_sent_total',
    help: 'Total number of messages sent',
    labelNames: ['type'],
    registers: [register]
  })
};

// 指标中间件
function metricsMiddleware(req, res, next) {
  const start = Date.now();
  const activeConnections = metrics.activeConnections;

  activeConnections.inc();

  res.on('finish', () => {
    const duration = (Date.now() - start) / 1000;

    metrics.httpRequestsTotal.inc({
      method: req.method,
      path: req.route?.path || req.path,
      status: res.statusCode
    });

    metrics.httpRequestDuration.observe({
      method: req.method,
      path: req.route?.path || req.path
    }, duration);

    activeConnections.dec();
  });

  next();
}

// 数据库查询指标
function recordDatabaseQuery(query, operation, duration) {
  metrics.databaseQueryDuration.observe({ query, operation }, duration);
}

// 缓存命中率指标
function recordCacheHitRate(cache, hitRate) {
  metrics.cacheHitRate.set({ cache }, hitRate);
}

// LLM调用指标
function recordLLMCall(model, duration, success) {
  metrics.llmCallsTotal.inc({ model, status: success ? 'success' : 'error' });
  metrics.llmCallDuration.observe({ model }, duration);
}

// 危机事件指标
function recordCrisisEvent(level) {
  metrics.crisisEventsTotal.inc({ level });
}

// 活跃用户指标
function setActiveUsers(count) {
  metrics.activeUsers.set(count);
}

// 消息发送指标
function recordMessageSent(type) {
  metrics.messagesSentTotal.inc({ type });
}

// 指标端点
app.get('/metrics', async (req, res) => {
  try {
    res.set('Content-Type', register.contentType);
    res.end(await register.metrics());
  } catch (error) {
    res.status(500).end(error);
  }
});

// 使用示例
app.use(metricsMiddleware);

// 记录数据库查询
const start = Date.now();
const result = await prisma.user.findMany();
recordDatabaseQuery('findManyUsers', 'read', (Date.now() - start) / 1000);

// 记录LLM调用
const llmStart = Date.now();
const llmResult = await llmService.generate(message);
recordLLMCall('qwen3-8b', (Date.now() - llmStart) / 1000, true);
```

### 2.3 告警规则

```yaml
# prometheus/rules/alerts.yml
groups:
  - name: service_alerts
    rules:
      # 服务宕机
      - alert: ServiceDown
        expr: up == 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "Service {{ $labels.job }} is down"
          description: "{{ $labels.job }} has been down for more than 1 minute"

      # 高错误率
      - alert: HighErrorRate
        expr: rate(http_requests_total{status=~"5.."}[5m]) / rate(http_requests_total[5m]) > 0.05
        for: 2m
        labels:
          severity: warning
        annotations:
          summary: "High error rate on {{ $labels.job }}"
          description: "Error rate is {{ $value | humanizePercentage }}"

      # 高延迟
      - alert: HighLatency
        expr: histogram_quantile(0.95, rate(http_request_duration_seconds_bucket[5m])) > 1
        for: 2m
        labels:
          severity: warning
        annotations:
          summary: "High latency on {{ $labels.job }}"
          description: "95th percentile latency is {{ $value }}s"

      # 高CPU使用率
      - alert: HighCPUUsage
        expr: 100 - (avg by(instance) (rate(node_cpu_seconds_total{mode="idle"}[5m])) * 100) > 80
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "High CPU usage on {{ $labels.instance }}"
          description: "CPU usage is {{ $value | humanizePercentage }}"

      # 高内存使用率
      - alert: HighMemoryUsage
        expr: (node_memory_MemTotal_bytes - node_memory_MemAvailable_bytes) / node_memory_MemTotal_bytes * 100 > 85
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "High memory usage on {{ $labels.instance }}"
          description: "Memory usage is {{ $value | humanizePercentage }}"

      # 磁盘空间不足
      - alert: DiskSpaceLow
        expr: (node_filesystem_avail_bytes{mountpoint="/"} / node_filesystem_size_bytes{mountpoint="/"}) * 100 < 10
        for: 5m
        labels:
          severity: critical
        annotations:
          summary: "Low disk space on {{ $labels.instance }}"
          description: "Disk space is {{ $value | humanizePercentage }}"

  - name: database_alerts
    rules:
      # PostgreSQL连接数过高
      - alert: PostgreSQLHighConnections
        expr: pg_stat_activity_count > 150
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "PostgreSQL high connections"
          description: "Connection count is {{ $value }}"

      # PostgreSQL复制延迟
      - alert: PostgreSQLReplicationLag
        expr: pg_replication_lag > 30
        for: 2m
        labels:
          severity: warning
        annotations:
          summary: "PostgreSQL replication lag"
          description: "Replication lag is {{ $value }}s"

      # Redis内存使用率过高
      - alert: RedisHighMemory
        expr: redis_memory_used_bytes / redis_memory_max_bytes > 0.8
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Redis high memory usage"
          description: "Memory usage is {{ $value | humanizePercentage }}"

      # Redis连接数过高
      - alert: RedisHighConnections
        expr: redis_connected_clients > 1000
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Redis high connections"
          description: "Connection count is {{ $value }}"

  - name: business_alerts
    rules:
      # 危机事件
      - alert: CrisisEventDetected
        expr: increase(crisis_events_total[5m]) > 0
        for: 0m
        labels:
          severity: critical
        annotations:
          summary: "Crisis event detected"
          description: "{{ $value }} crisis events in the last 5 minutes"

      # LLM服务不可用
      - alert: LLMServiceUnavailable
        expr: rate(llm_calls_total{status="error"}[5m]) / rate(llm_calls_total[5m]) > 0.5
        for: 2m
        labels:
          severity: critical
        annotations:
          summary: "LLM service unavailable"
          description: "LLM error rate is {{ $value | humanizePercentage }}"

      # LLM响应延迟过高
      - alert: LLMHighLatency
        expr: histogram_quantile(0.95, rate(llm_call_duration_seconds_bucket[5m])) > 10
        for: 2m
        labels:
          severity: warning
        annotations:
          summary: "LLM high latency"
          description: "95th percentile latency is {{ $value }}s"

      # 活跃用户数异常
      - alert: ActiveUsersAnomaly
        expr: active_users < 10 or active_users > 10000
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Active users anomaly"
          description: "Active users count is {{ $value }}"
```

```javascript
    return (req, res, next) => {
      const start = Date.now();

      // 添加请求ID到请求对象
      req.requestId = requestId;
      
      res.on('finish', () => {

        this.info('HTTP Request', {
          requestId,
          method: req.method,
          url: req.url,
          status: res.statusCode,
          duration,
          userAgent: req.get('User-Agent'),
          ip: req.ip,
          userId: req.user?.userId
        });
      });
      
      next();
    };
  }

  // 错误日志中间件
  errorLogger() {
    return (err, req, res, next) => {
      this.error('Unhandled Error', {
        requestId: req.requestId,
        error: err.message,
        stack: err.stack,
        method: req.method,
        url: req.url,
        userId: req.user?.userId
      });
      
      res.status(500).json({
        error: 'Internal Server Error',
        requestId: req.requestId
      });
    };

      action,
      ...data
    });
  }

  securityLog(event, data) {
    this.warn('Security Event', {

  }
```

### 6.2 链路追踪实现

```javascript
// 链路追踪服务
const { initTracer } = require('jaeger-client');
const opentracing = require('opentracing');

class TracingService {
  constructor(serviceName) {
    this.tracer = initTracer({
      serviceName,
      sampler: {
        type: 'const',
        param: 1
      },
      reporter: {
        logSpans: true,
        agentHost: process.env.JAEGER_AGENT_HOST || 'jaeger-agent',
        agentPort: process.env.JAEGER_AGENT_PORT || 6832
      }
    });
    
    opentracing.initGlobalTracer(this.tracer);
  }

  // 创建span
  startSpan(operationName, parentSpan = null) {
    const options = {};
    
    if (parentSpan) {
      options.childOf = parentSpan;
    }
    
    return this.tracer.startSpan(operationName, options);
  }

  // 中间件
  middleware() {
    return (req, res, next) => {
      const span = this.startSpan('http_request');
      
      span.setTag('http.method', req.method);
      span.setTag('http.url', req.url);
      span.setTag('http.request_id', req.requestId);
      
      req.span = span;
      
      res.on('finish', () => {
        span.setTag('http.status_code', res.statusCode);
        
        if (res.statusCode >= 400) {
          span.setTag('error', true);
        }
        
        span.finish();
      });
      
      next();
    };
  }

  // 追踪数据库查询
  async traceDatabaseQuery(query, operation, fn) {
    const parentSpan = opentracing.globalTracer().startSpan('database_query');
    
    parentSpan.setTag('db.query', query);
    parentSpan.setTag('db.operation', operation);
    
    try {
      const result = await fn();
      parentSpan.setTag('db.success', true);
      return result;
    } catch (error) {
      parentSpan.setTag('error', true);
      parentSpan.setTag('error.message', error.message);
      throw error;
    } finally {
      parentSpan.finish();
    }
  }

  // 追踪LLM调用
  async traceLLMCall(model, operation, fn) {
    const parentSpan = opentracing.globalTracer().startSpan('llm_call');
    
    parentSpan.setTag('llm.model', model);
    parentSpan.setTag('llm.operation', operation);
    
    try {
      const result = await fn();
      parentSpan.setTag('llm.success', true);
      return result;
    } catch (error) {
      parentSpan.setTag('error', true);
      parentSpan.setTag('error.message', error.message);
      throw error;
    } finally {
      parentSpan.finish();
    }
  }

  // 追踪服务调用
  async traceServiceCall(serviceName, operation, fn) {
    const parentSpan = opentracing.globalTracer().startSpan('service_call');
    
    parentSpan.setTag('service.name', serviceName);
    parentSpan.setTag('service.operation', operation);
    
    try {
      const result = await fn();
      parentSpan.setTag('service.success', true);
      return result;
    } catch (error) {
      parentSpan.setTag('error', true);
      parentSpan.setTag('error.message', error.message);
      throw error;
    } finally {
      parentSpan.finish();
    }
  }

  // 关闭追踪器
  close() {
    this.tracer.close();
  }
}

// 使用示例
const tracingService = new TracingService('user-service');

// 中间件
app.use(tracingService.middleware());

// 追踪数据库查询
const user = await tracingService.traceDatabaseQuery(
  'findUserById',
  'read',
  () => prisma.user.findUnique({ where: { id: userId } })
);

// 追踪LLM调用
const response = await tracingService.traceLLMCall(
  'qwen3-8b',
  'generate',
  () => llmService.generate(message)
);
```
