### 2.3 读写分离实现

```javascript
// Prisma 读写分离配置
const { PrismaClient } = require('@prisma/client');

class DatabaseManager {
  constructor() {
    // 主库 - 用于写操作
    this.primary = new PrismaClient({
      datasources: {
        db: {
          url: process.env.DATABASE_URL
        }
      },
      log: ['query', 'info', 'warn', 'error']
    });

    // 从库 - 用于读操作
    this.replica = new PrismaClient({
      datasources: {
        db: {
          url: process.env.DATABASE_REPLICA_URL
        }
      },
      log: ['query', 'info', 'warn', 'error']
    });
  }

  // 写操作 - 使用主库
  async write(operation) {
    return await operation(this.primary);
  }

  // 读操作 - 使用从库
  async read(operation) {
    return await operation(this.replica);
  }

  // 读写操作 - 使用主库
  async readWrite(readOp, writeOp) {
    const readResult = await readOp(this.primary);
    const writeResult = await writeOp(this.primary);
    return { readResult, writeResult };
  }

  // 事务 - 使用主库
  async transaction(operations) {
    return await this.primary.$transaction(async (prisma) => {
      const results = [];
      for (const op of operations) {
        results.push(await op(prisma));
      }
      return results;
    });
  }

  // 关闭连接
  async disconnect() {
    await this.primary.$disconnect();
    await this.replica.$disconnect();
  }
}

// 使用示例
const db = new DatabaseManager();

// 写操作
const user = await db.write(async (prisma) => {
  return await prisma.user.create({
    data: { phone: '13800138000' }
  });
});

// 读操作
const users = await db.read(async (prisma) => {
  return await prisma.user.findMany({
    where: { isActive: true }
  });
});

// 事务
const result = await db.transaction([
  async (prisma) => {
    return await prisma.user.update({

    });

  async (prisma) => {
    return await prisma.loginLog.create({

  }
]);
```

### 2.5 分区表设计

```sql
-- 消息表按月分区
CREATE TABLE messages (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    conversation_id UUID NOT NULL,
    role VARCHAR(10) NOT NULL,
    content TEXT NOT NULL,
    emotion VARCHAR(20),
    importance INTEGER DEFAULT 3,
    created_at TIMESTAMPTZ DEFAULT now()
) PARTITION BY RANGE (created_at);

-- 创建分区
CREATE TABLE messages_2026_01 PARTITION OF messages
    FOR VALUES FROM ('2026-01-01') TO ('2026-02-01');

CREATE TABLE messages_2026_02 PARTITION OF messages
    FOR VALUES FROM ('2026-02-01') TO ('2026-03-01');

-- 自动创建分区函数
CREATE OR REPLACE FUNCTION create_monthly_partition()
RETURNS void AS $$
DECLARE
    next_month DATE;
    partition_name TEXT;
    start_date DATE;
    end_date DATE;
BEGIN
    next_month := date_trunc('month', CURRENT_DATE + INTERVAL '1 month');
    partition_name := 'messages_' || to_char(next_month, 'YYYY_MM');
    start_date := to_char(next_month, 'YYYY-MM-DD');
    end_date := to_char(next_month + INTERVAL '1 month', 'YYYY-MM-DD');
    
    EXECUTE format('CREATE TABLE IF NOT EXISTS %I PARTITION OF messages FOR VALUES FROM (%L) TO (%L)',
                   partition_name, start_date, end_date);
END;
$$ LANGUAGE plpgsql;

-- 定时任务：每月创建新分区
SELECT cron.schedule('create-partition', '0 0 1 * *', 'SELECT create_monthly_partition()');
```

```javascript
// Redis 缓存管理器
class RedisCacheManager {
  constructor() {
    this.redis = new Redis.Cluster([
      { host: 'redis-node-1', port: 6379 },
      { host: 'redis-node-2', port: 6379 },
      { host: 'redis-node-3', port: 6379 }
    ], {
      redisOptions: {
        password: process.env.REDIS_PASSWORD
      },
      scaleReads: 'slave'
    });
  }

  // 用户缓存
  async getUser(userId) {
    const key = `user:${userId}`;
    return await this.get(key);
  }

  async setUser(userId, data, ttl = 3600) {
    const key = `user:${userId}`;
    await this.set(key, data, ttl);
  }

  // 会话缓存
  async getConversation(conversationId) {
    const key = `conversation:${conversationId}`;
    return await this.get(key);
  }

  async setConversation(conversationId, data, ttl = 1800) {
    const key = `conversation:${conversationId}`;
    await this.set(key, data, ttl);
  }

  // 对话上下文缓存
  async getConversationContext(conversationId) {
    const key = `context:${conversationId}`;
    return await this.get(key);
  }

  async setConversationContext(conversationId, context, ttl = 3600) {
    const key = `context:${conversationId}`;
    await this.set(key, context, ttl);
  }

  // 记忆缓存
  async getUserMemories(userId) {
    const key = `memories:${userId}`;
    return await this.get(key);
  }

  async setUserMemories(userId, memories, ttl = 1800) {
    const key = `memories:${userId}`;
    await this.set(key, memories, ttl);
  }

  // 热点数据缓存
  async getHotData(key) {
    return await this.get(`hot:${key}`);
  }

  async setHotData(key, data, ttl = 300) {
    await this.set(`hot:${key}`, data, ttl);
  }

  // 通用缓存操作
  async get(key) {
    try {
      const data = await this.redis.get(key);
      return data ? JSON.parse(data) : null;
    } catch (error) {
      console.error('Redis get error:', error);
      return null;
    }
  }

  async set(key, value, ttl = 3600) {
    try {
      await this.redis.setex(key, ttl, JSON.stringify(value));
    } catch (error) {
      console.error('Redis set error:', error);
    }
  }

  async delete(key) {
    try {
      await this.redis.del(key);
    } catch (error) {
      console.error('Redis delete error:', error);
    }
  }

  // 批量操作
  async mget(keys) {
    try {
      const values = await this.redis.mget(keys);
      return values.map(v => v ? JSON.parse(v) : null);
    } catch (error) {
      console.error('Redis mget error:', error);
      return keys.map(() => null);
    }
  }

  async mset(keyValuePairs, ttl = 3600) {
    try {
      const pipeline = this.redis.pipeline();
      for (const [key, value] of Object.entries(keyValuePairs)) {
        pipeline.setex(key, ttl, JSON.stringify(value));
      }
      await pipeline.exec();
    } catch (error) {
      console.error('Redis mset error:', error);
    }
  }

  // 增量操作
  async increment(key, delta = 1) {
    try {
      return await this.redis.incrby(key, delta);
    } catch (error) {
      console.error('Redis increment error:', error);
      return null;
    }
  }

  // 过期时间管理
  async getTTL(key) {
    try {
      return await this.redis.ttl(key);
    } catch (error) {
      console.error('Redis getTTL error:', error);
      return -1;
    }
  }

  async setTTL(key, ttl) {
    try {
      await this.redis.expire(key, ttl);
    } catch (error) {
      console.error('Redis setTTL error:', error);
    }
  }
}
```

### 3.4 消息队列

```javascript
// Redis Streams 消息队列
class RedisMessageQueue {
  constructor() {
    this.redis = new Redis.Cluster([
      { host: 'redis-node-1', port: 6379 },
      { host: 'redis-node-2', port: 6379 },
      { host: 'redis-node-3', port: 6379 }
    ]);
  }

  // 发布消息
  async publish(stream, message, id = '*') {
    try {
      return await this.redis.xadd(stream, id, 'data', JSON.stringify(message));
    } catch (error) {
      console.error('Redis publish error:', error);
      throw error;
    }
  }

  // 创建消费者组
  async createGroup(stream, group) {
    try {
      await this.redis.xgroup('CREATE', stream, group, '0', 'MKSTREAM');
    } catch (error) {
      if (!error.message.includes('BUSYGROUP')) {
        console.error('Redis createGroup error:', error);
        throw error;
      }
    }
  }

  // 消费消息
  async subscribe(stream, group, consumer, handler, count = 10) {
    await this.createGroup(stream, group);

    while (true) {
      try {
        const messages = await this.redis.xreadgroup(
          'GROUP', group, consumer,
          'COUNT', count,
          'BLOCK', 2000,
          'STREAMS', stream, '>'
        );

        if (messages) {
          for (const [streamName, entries] of messages) {
            for (const [id, fields] of entries) {
              try {
                const data = JSON.parse(fields.data);
                await handler(data, id);
                await this.redis.xack(streamName, group, id);
              } catch (error) {
                console.error('Message processing error:', error);
                // 可以将失败消息放入死信队列
                await this.publish(`${stream}:dead`, {
                  originalId: id,
                  error: error.message,
                  data: fields.data
                });
              }
            }
          }
        }
      } catch (error) {
        console.error('Redis subscribe error:', error);
        await new Promise(resolve => setTimeout(resolve, 1000));
      }
    }
  }

  // 获取消息
  async getMessages(stream, start = '-', end = '+', count = 100) {
    try {
      const messages = await this.redis.xrange(stream, start, end, 'COUNT', count);
      return messages.map(([id, fields]) => ({
        id,
        data: JSON.parse(fields.data)
      }));
    } catch (error) {
      console.error('Redis getMessages error:', error);
      return [];
    }
  }

  // 删除消息
  async deleteMessage(stream, id) {
    try {
      await this.redis.xdel(stream, id);
    } catch (error) {
      console.error('Redis deleteMessage error:', error);
    }
  }

  // 获取流信息
  async getStreamInfo(stream) {
    try {
      return await this.redis.xinfo('STREAM', stream);
    } catch (error) {
      console.error('Redis getStreamInfo error:', error);
      return null;
    }
  }
}

// 使用示例
const mq = new RedisMessageQueue();

// 发布对话消息
await mq.publish('chat-messages', {
  userId: 'user-123',
  conversationId: 'conv-456',
  message: '用户消息内容',
  timestamp: new Date().toISOString()
});

// 消费对话消息
await mq.subscribe('chat-messages', 'memory-group', 'memory-consumer', async (message, id) => {
  // 提取记忆
  const memories = await memoryService.extractMemories(message.message);
  
  // 存储记忆
  for (const memory of memories) {
    await memoryService.saveMemory(message.userId, memory);
  }
});
```

```python
from pymilvus import Collection, connections
import numpy as np

class MilvusService:
    def __init__(self):
        connections.connect(
            alias="default",
            host="milvus-service",
            port="19530"
        )
        self.collection = Collection("memory_vectors")
        self.collection.load()
    
    # 插入向量
    def insert_vectors(self, data):
        """
        data: list of dicts with keys: id, user_id, memory_id, content, embedding, importance, created_at
        """
        try:
            result = self.collection.insert(data)
            return result
        except Exception as e:
            print(f"Insert error: {e}")
            raise
    
    # 搜索相似向量
    def search_similar(self, user_id, query_embedding, top_k=10, threshold=0.7):
        """
        搜索与查询向量最相似的记忆
        """
        try:
            search_params = {
                "metric_type": "COSINE",
                "params": {"ef": 100}
            }
            
            results = self.collection.search(
                data=[query_embedding],
                anns_field="embedding",
                param=search_params,
                limit=top_k,
                expr=f'user_id == "{user_id}"',
                output_fields=["memory_id", "content", "importance"]
            )
            
            # 过滤低相似度结果
            filtered_results = []
            for hit in results[0]:
                if hit.score >= threshold:
                    filtered_results.append({
                        "memory_id": hit.entity.get("memory_id"),
                        "content": hit.entity.get("content"),
                        "importance": hit.entity.get("importance"),
                        "similarity": hit.score
                    })
            
            return filtered_results
        except Exception as e:
            print(f"Search error: {e}")
            raise
    
    # 删除向量
    def delete_vectors(self, ids):
        try:
            self.collection.delete(f'id in {ids}')
        except Exception as e:
            print(f"Delete error: {e}")
            raise
    
    # 更新向量
    def update_vector(self, id, data):
        try:
            self.collection.delete(f'id == "{id}"')
            self.collection.insert([data])
        except Exception as e:
            print(f"Update error: {e}")
            raise

                expr=f'user_id == "{user_id}"',

                limit=limit
            )

        except Exception as e:

            raise

            self.collection.delete(f'user_id == "{user_id}"')
        except Exception as e:
```

```javascript
// 最终一致性管理器
class EventualConsistencyManager {
  constructor() {
    this.redis = new Redis();
    this.neo4j = new Neo4jDriver();
    this.milvus = new MilvusService();
  }

  // 更新记忆 - 最终一致性
  async updateMemory(userId, memoryId, updates) {
    // 1. 更新PostgreSQL (主数据源)
    const memory = await prisma.memory.update({
      where: { id: memoryId },
      data: updates
    });

    // 2. 发布更新事件
    await this.redis.publish('memory-updated', JSON.stringify({
      userId,
      memoryId,
      updates,
      timestamp: Date.now()
    }));

    return memory;
  }

  // 处理记忆更新事件
  async handleMemoryUpdated(event) {
    const { userId, memoryId, updates } = JSON.parse(event);

    // 3. 更新Redis缓存
    await this.updateRedisCache(userId, memoryId, updates);

    // 4. 更新Neo4j知识图谱
    await this.updateNeo4j(userId, memoryId, updates);

    // 5. 更新Milvus向量
    await this.updateMilvus(userId, memoryId, updates);
  }

  // 更新Redis缓存
  async updateRedisCache(userId, memoryId, updates) {
    const cacheKey = `memories:${userId}`;
    const cached = await this.redis.get(cacheKey);
    
    if (cached) {
      const memories = JSON.parse(cached);
      const index = memories.findIndex(m => m.id === memoryId);
      
      if (index !== -1) {
        memories[index] = { ...memories[index], ...updates };
        await this.redis.setex(cacheKey, 1800, JSON.stringify(memories));
      }
    }
  }

  // 更新Neo4j
  async updateNeo4j(userId, memoryId, updates) {
    const session = this.neo4j.driver.session();
    try {
      await session.run(
        `MATCH (m:Memory {id: $memoryId})
         SET m += $updates
         RETURN m`,
        { memoryId, updates }
      );
    } finally {
      await session.close();
    }
  }

  // 更新Milvus
  async updateMilvus(userId, memoryId, updates) {
    if (updates.content) {
      // 重新生成向量
      const embedding = await this.generateEmbedding(updates.content);
      
      await this.milvus.update_vector(`vec_${memoryId}`, {
        id: `vec_${memoryId}`,
        user_id: userId,
        memory_id: memoryId,
        content: updates.content,
        embedding: embedding,
        importance: updates.importance || 5,
        created_at: Date.now()
      });
    }
  }

  // 生成向量嵌入
  async generateEmbedding(text) {
    // 调用向量生成服务
    const response = await axios.post('http://embedding-service/api/embed', {
      text
    });
    return response.data.embedding;
  }
}

// 使用示例

  importance: 9
});

  await consistencyManager.handleMemoryUpdated(event);
});
```

### 6.2 分布式事务 (Saga模式)

```javascript
// Saga 编排器
class SagaOrchestrator {
  constructor() {
    this.steps = [];
    this.compensations = [];
  }

  addStep(execute, compensate) {
    this.steps.push(execute);
    this.compensations.push(compensate);
  }

  async execute(context) {
    const executedSteps = [];

    for (let i = 0; i < this.steps.length; i++) {
      try {
        await this.steps[i](context);
        executedSteps.push(i);
      } catch (error) {
        // 执行补偿操作
        for (let j = executedSteps.length - 1; j >= 0; j--) {
          try {
            await this.compensations[executedSteps[j]](context);
          } catch (compensationError) {
            console.error('Compensation failed:', compensationError);
          }
        }
        throw error;
      }
    }
  }
}

        content: ctx.content,
        type: ctx.type,

  // 补偿1：删除PostgreSQL记忆

    if (ctx.memory) {
      await prisma.memory.delete({ where: { id: ctx.memory.id } });
    }
  }

        await session.run(

        await session.close();
      }
    }
  }
);
```
