## 一、容器化架构

### 1.1 Docker镜像设计

```dockerfile
# 多阶段构建 - Node.js应用
# 阶段1：构建阶段
FROM node:20-alpine AS builder

WORKDIR /app

# 复制依赖文件
COPY package*.json ./

# 安装依赖
RUN npm ci --only=production

# 复制源代码
COPY . .

# 构建应用
RUN npm run build

# 阶段2：生产阶段
FROM node:20-alpine AS production

WORKDIR /app

# 安装生产依赖
COPY --from=builder /app/node_modules ./node_modules
COPY --from=builder /app/dist ./dist
COPY --from=builder /app/package.json ./

# 创建非root用户
RUN addgroup -g 1001 -S nodejs
RUN adduser -S nodejs -u 1001

# 更改文件所有权
RUN chown -R nodejs:nodejs /app

# 切换到非root用户
USER nodejs

# 暴露端口
EXPOSE 3000

# 健康检查
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \
  CMD node healthcheck.js

# 启动应用
CMD ["node", "dist/app.js"]
```

name: CI/CD Pipeline

on:

    - name: Run linting

    - name: Run tests

    - name: Run coverage

### 3.2 部署脚本

```bash
#!/bin/bash
# deploy.sh

set -e

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

# 日志函数
log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# 检查依赖
check_dependencies() {
    log_info "Checking dependencies..."
    
    if ! command -v kubectl &> /dev/null; then
        log_error "kubectl is not installed"
        exit 1
    fi
    
    if ! command -v docker &> /dev/null; then
        log_error "docker is not installed"
        exit 1
    fi
    
    log_info "All dependencies are installed"
}

# 构建镜像
build_images() {
    log_info "Building Docker images..."
    
    # 构建用户服务
    docker build -t cyber-sister/user-service:latest ./services/user-service
    
    # 构建对话服务
    docker build -t cyber-sister/chat-service:latest ./services/chat-service
    
    # 构建记忆服务
    docker build -t cyber-sister/memory-service:latest ./services/memory-service
    
    # 构建LLM服务
    docker build -t cyber-sister/llm-service:latest ./services/llm-service
    
    log_info "Docker images built successfully"
}

# 推送镜像
push_images() {
    log_info "Pushing Docker images..."
    
    docker push cyber-sister/user-service:latest
    docker push cyber-sister/chat-service:latest
    docker push cyber-sister/memory-service:latest
    docker push cyber-sister/llm-service:latest
    
    log_info "Docker images pushed successfully"
}

# 部署到Kubernetes
deploy_to_kubernetes() {
    local environment=$1
    
    log_info "Deploying to $environment..."
    
    # 创建命名空间
    kubectl apply -f k8s/namespace.yaml
    
    # 应用配置
    kubectl apply -f k8s/configmap.yaml -n cyber-sister-$environment
    kubectl apply -f k8s/secret.yaml -n cyber-sister-$environment
    
    # 部署服务
    kubectl apply -f k8s/user-service-deployment.yaml -n cyber-sister-$environment
    kubectl apply -f k8s/chat-service-deployment.yaml -n cyber-sister-$environment
    kubectl apply -f k8s/memory-service-deployment.yaml -n cyber-sister-$environment
    kubectl apply -f k8s/llm-service-deployment.yaml -n cyber-sister-$environment
    
    # 等待部署完成
    kubectl rollout status deployment/user-service -n cyber-sister-$environment
    kubectl rollout status deployment/chat-service -n cyber-sister-$environment
    kubectl rollout status deployment/memory-service -n cyber-sister-$environment
    kubectl rollout status deployment/llm-service -n cyber-sister-$environment
    
    log_info "Deployment to $environment completed successfully"
}

# 回滚部署
rollback() {
    local environment=$1
    local service=$2
    
    log_warn "Rolling back $service in $environment..."
    
    kubectl rollout undo deployment/$service -n cyber-sister-$environment
    
    log_info "Rollback completed"
}

# 主函数
main() {
    local environment=${1:-staging}
    
    check_dependencies
    build_images
    push_images
    deploy_to_kubernetes $environment
}

# 执行主函数
main "$@"
```

## 六、备份与恢复

### 6.1 自动备份脚本

```bash
#!/bin/bash
# backup.sh

set -e

# 配置
BACKUP_DIR="/backup"
DATE=$(date +%Y%m%d_%H%M%S)
RETENTION_DAYS=7

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

# 日志函数
log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# 创建备份目录
create_backup_dir() {
    mkdir -p $BACKUP_DIR/postgresql
    mkdir -p $BACKUP_DIR/redis
    mkdir -p $BACKUP_DIR/neo4j
    mkdir -p $BACKUP_DIR/milvus
}

# PostgreSQL备份
backup_postgresql() {
    log_info "Starting PostgreSQL backup..."
    
    pg_dump -h postgres -U postgres -d cyber_sister -F c -f $BACKUP_DIR/postgresql/full_$DATE.dump
    
    log_info "PostgreSQL backup completed"
}

# Redis备份
backup_redis() {
    log_info "Starting Redis backup..."
    
    redis-cli -h redis -a $REDIS_PASSWORD BGSAVE
    sleep 10
    cp /var/lib/redis/dump.rdb $BACKUP_DIR/redis/dump_$DATE.rdb
    
    log_info "Redis backup completed"
}

# Neo4j备份
backup_neo4j() {
    log_info "Starting Neo4j backup..."
    
    neo4j-admin database backup --to-path=$BACKUP_DIR/neo4j/neo4j_$DATE neo4j
    
    log_info "Neo4j backup completed"
}

# Milvus备份
backup_milvus() {
    log_info "Starting Milvus backup..."
    
    milvus-backup create -n backup_$DATE
    
    log_info "Milvus backup completed"
}

# 清理旧备份
cleanup_old_backups() {
    log_info "Cleaning up old backups..."
    
    find $BACKUP_DIR -name "*.dump" -mtime +$RETENTION_DAYS -delete
    find $BACKUP_DIR -name "*.rdb" -mtime +$RETENTION_DAYS -delete
    find $BACKUP_DIR -name "neo4j_*" -mtime +$RETENTION_DAYS -exec rm -rf {} \;
    find $BACKUP_DIR -name "backup_*" -mtime +$RETENTION_DAYS -exec rm -rf {} \;
    
    log_info "Old backups cleaned up"
}

# 上传到云存储
upload_to_cloud() {
    log_info "Uploading backups to cloud storage..."
    
    aws s3 sync $BACKUP_DIR s3://cyber-sister-backup/ --exclude "*.tmp"
    
    log_info "Backups uploaded to cloud storage"
}

# 主函数
main() {
    log_info "Starting backup process..."
    
    create_backup_dir
    backup_postgresql
    backup_redis
    backup_neo4j
    backup_milvus
    cleanup_old_backups
    upload_to_cloud
    
    log_info "Backup process completed successfully"
}

# 执行主函数
main "$@"
```

### 6.2 恢复脚本

```bash
#!/bin/bash
# restore.sh

set -e

# 配置
BACKUP_DIR="/backup"

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

# 日志函数
log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# 恢复PostgreSQL
restore_postgresql() {
    local backup_file=$1
    
    log_info "Restoring PostgreSQL from $backup_file..."
    
    pg_restore -h postgres -U postgres -d cyber_sister -c $backup_file
    
    log_info "PostgreSQL restoration completed"
}

# 恢复Redis
restore_redis() {
    local backup_file=$1
    
    log_info "Restoring Redis from $backup_file..."
    
    redis-cli -h redis -a $REDIS_PASSWORD SHUTDOWN NOSAVE
    cp $backup_file /var/lib/redis/dump.rdb
    redis-server --daemonize yes
    
    log_info "Redis restoration completed"
}

# 恢复Neo4j
restore_neo4j() {
    local backup_dir=$1
    
    log_info "Restoring Neo4j from $backup_dir..."
    
    neo4j stop
    neo4j-admin database restore --from-path=$backup_dir neo4j
    neo4j start
    
    log_info "Neo4j restoration completed"
}

# 列出可用备份
list_backups() {
    log_info "Available backups:"
    
    echo "PostgreSQL backups:"
    ls -la $BACKUP_DIR/postgresql/
    
    echo "Redis backups:"
    ls -la $BACKUP_DIR/redis/
    
    echo "Neo4j backups:"
    ls -la $BACKUP_DIR/neo4j/
    
    echo "Milvus backups:"
    ls -la $BACKUP_DIR/milvus/
}

# 主函数
main() {
    local action=$1
    local backup_file=$2
    
    case $action in
        list)
            list_backups
            ;;
        postgresql)
            restore_postgresql $backup_file
            ;;
        redis)
            restore_redis $backup_file
            ;;
        neo4j)
            restore_neo4j $backup_file
            ;;
        *)
            echo "Usage: $0 {list|postgresql|redis|neo4j} [backup_file]"
            exit 1
            ;;
    esac
}

# 执行主函数
main "$@"
```
