## 一、架构设计原则

### 1.1 核心设计原则

| 原则 | 描述 | 实施策略 |
|------|------|----------|
| **服务自治** | 每个微服务独立开发、部署、扩展 | 独立数据库、独立进程、独立部署 |
| **松耦合** | 服务间通过API和消息队列通信 | 异步消息、事件驱动、API网关 |
| **高内聚** | 相关功能聚合在同一服务内 | 按业务领域划分服务 |
| **故障隔离** | 单个服务故障不影响整体系统 | 熔断器、舱壁模式、超时控制 |
| **数据一致性** | 保证分布式环境下的数据一致性 | 最终一致性、Saga模式、分布式事务 |
| **可观测性** | 系统运行状态可监控、可追踪 | 链路追踪、指标监控、日志聚合 |

### 1.2 技术选型原则

| 维度 | 选型 | 理由 |
|------|------|------|
| **服务框架** | Node.js + Express | 保持与现有技术栈一致，降低迁移成本 |
| **服务通信** | REST API + gRPC | REST用于外部API，gRPC用于内部服务通信 |
| **消息队列** | Redis Streams | 轻量级、高性能、支持消息持久化 |
| **服务发现** | Consul | 成熟的服务发现和配置管理 |
| **容器化** | Docker + Docker Compose | 标准化部署，支持水平扩展 |
| **负载均衡** | Nginx + Kubernetes Ingress | 七层负载均衡，支持健康检查 |
| **监控告警** | Prometheus + Grafana | 开源、可视化、告警能力强 |
| **链路追踪** | Jaeger | 分布式链路追踪，支持性能分析 |

### 3.2 故障恢复机制

#### 3.2.1 熔断器模式

```javascript
// 熔断器实现
class CircuitBreaker {
  constructor(options = {}) {
    this.failureThreshold = options.failureThreshold || 5;
    this.resetTimeout = options.resetTimeout || 60000;
    this.state = 'CLOSED'; // CLOSED, OPEN, HALF_OPEN
    this.failureCount = 0;
    this.lastFailureTime = null;
  }

  async execute(fn) {
    if (this.state === 'OPEN') {
      if (Date.now() - this.lastFailureTime > this.resetTimeout) {
        this.state = 'HALF_OPEN';
      } else {
        throw new Error('Circuit breaker is OPEN');
      }
    }

    try {
      const result = await fn();
      this.onSuccess();
      return result;
    } catch (error) {
      this.onFailure();
      throw error;
    }
  }

  onSuccess() {
    this.failureCount = 0;
    this.state = 'CLOSED';
  }

  onFailure() {
    this.failureCount++;
    this.lastFailureTime = Date.now();
    if (this.failureCount >= this.failureThreshold) {
      this.state = 'OPEN';
    }
  }
}

// 使用示例
const llmCircuitBreaker = new CircuitBreaker({
  failureThreshold: 5,
  resetTimeout: 60000
});

async function callLLMService(message) {
  return llmCircuitBreaker.execute(async () => {
    const response = await axios.post('http://llm-service/api/llm/generate', {
      message,
      timeout: 10000
    });
    return response.data;
  });
}
```

#### 3.2.2 舱壁模式

```javascript
// 舱壁模式 - 为不同服务分配独立的连接池
const servicePools = {
  llm: new Pool({ max: 10, min: 2 }),
  memory: new Pool({ max: 20, min: 5 }),
  user: new Pool({ max: 15, min: 3 })
};

async function callService(serviceName, request) {
  const pool = servicePools[serviceName];
  if (!pool) {
    throw new Error(`Unknown service: ${serviceName}`);
  }
  
  return pool.acquire().then(async (connection) => {
    try {
      return await connection.execute(request);
    } finally {
      pool.release(connection);
    }
  });
}
```

```javascript
// 超时控制中间件
function timeoutMiddleware(timeout = 5000) {
  return (req, res, next) => {
    const timeoutId = setTimeout(() => {
      res.status(408).json({
        error: 'Request timeout',
        code: 'TIMEOUT'
      });
    }, timeout);

    res.on('finish', () => {
      clearTimeout(timeoutId);
    });

    next();
  };
}
```

### 4.2 缓存策略

#### 4.2.1 多级缓存架构

```javascript
// 多级缓存实现
class MultiLevelCache {
  constructor() {
    this.l1Cache = new Map(); // 本地缓存
    this.l2Cache = redis;     // Redis缓存
    this.l3Cache = prisma;    // 数据库
  }

  async get(key) {
    // L1: 本地缓存
    if (this.l1Cache.has(key)) {
      return this.l1Cache.get(key);
    }

    // L2: Redis缓存
    const cached = await this.l2Cache.get(key);
    if (cached) {
      this.l1Cache.set(key, JSON.parse(cached));
      return JSON.parse(cached);
    }

    // L3: 数据库
    const data = await this.l3Cache.user.findUnique({ where: { id: key } });
    if (data) {
      await this.l2Cache.setex(key, 3600, JSON.stringify(data));
      this.l1Cache.set(key, data);
    }

    return data;
  }

  async set(key, value, ttl = 3600) {
    this.l1Cache.set(key, value);
    await this.l2Cache.setex(key, ttl, JSON.stringify(value));
  }

  async invalidate(key) {
    this.l1Cache.delete(key);
    await this.l2Cache.del(key);
  }
}
```

```javascript
// Cache-Aside 模式
async function getUserWithCache(userId) {
  // 1. 先查缓存
  const cached = await cache.get(`user:${userId}`);
  if (cached) {
    return cached;
  }

  // 2. 缓存未命中，查数据库
  const user = await prisma.user.findUnique({
    where: { id: userId }
  });

  if (user) {
    // 3. 写入缓存
    await cache.set(`user:${userId}`, user, 3600);
  }

  return user;
}

// 写入时更新缓存
async function updateUser(userId, data) {
  // 1. 更新数据库
  const user = await prisma.user.update({
    where: { id: userId },
    data
  });

  // 2. 更新缓存
  await cache.set(`user:${userId}`, user, 3600);

  // 3. 发布更新事件
  await redis.publish('user-updated', JSON.stringify({ userId, data }));

  return user;
}
```

### 4.3 分布式事务

#### 4.3.1 Saga 模式

```javascript
// Saga 编排器
class SagaOrchestrator {
  constructor() {
    this.steps = [];
    this.compensations = [];
  }

  addStep(execute, compensate) {
    this.steps.push(execute);
    this.compensations.push(compensate);
  }

  async execute(context) {
    const executedSteps = [];

    for (let i = 0; i < this.steps.length; i++) {
      try {
        await this.steps[i](context);
        executedSteps.push(i);
      } catch (error) {
        // 执行补偿操作
        for (let j = executedSteps.length - 1; j >= 0; j--) {
          try {
            await this.compensations[executedSteps[j]](context);
          } catch (compensationError) {
            console.error('Compensation failed:', compensationError);
          }
        }
        throw error;
      }
    }
  }
}

// 使用示例：用户注册Saga
const userRegistrationSaga = new SagaOrchestrator();
```

```javascript
// Redis Streams 消息队列
class MessageQueue {
  constructor() {
    this.redis = new Redis();
  }

  // 发布消息
  async publish(stream, message) {
    await this.redis.xadd(stream, '*', 'data', JSON.stringify(message));
  }

  // 消费消息
  async subscribe(stream, group, consumer, handler) {
    while (true) {
      const messages = await this.redis.xreadgroup(
        'GROUP', group, consumer,
        'COUNT', 10,
        'BLOCK', 2000,
        'STREAMS', stream, '>'
      );

      if (messages) {
        for (const [stream, entries] of messages) {
          for (const [id, fields] of entries) {
            try {
              await handler(JSON.parse(fields.data));
              await this.redis.xack(stream, group, id);
            } catch (error) {
              console.error('Message processing failed:', error);
            }
          }
        }
      }
    }
  }
}

// 使用示例
const mq = new MessageQueue();

// 发布对话消息
await mq.publish('chat-messages', {
  userId: 'user-123',
  conversationId: 'conv-456',
  message: '用户消息内容',
  timestamp: new Date().toISOString()
});

// 消费对话消息
await mq.subscribe('chat-messages', 'memory-group', 'memory-consumer', async (message) => {
  // 提取记忆
  const memories = await memoryService.extractMemories(message.message);
  
  // 存储记忆
  for (const memory of memories) {
    await memoryService.saveMemory(message.userId, memory);
  }
});
```

```javascript
// 数据加密工具
const crypto = require('crypto');

class EncryptionService {
  constructor() {
    this.algorithm = 'aes-256-gcm';
    this.secretKey = process.env.ENCRYPTION_KEY;
  }

  encrypt(text) {
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipher(this.algorithm, this.secretKey);
    
    let encrypted = cipher.update(text, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    
    const authTag = cipher.getAuthTag();
    
    return {
      iv: iv.toString('hex'),
      encrypted,
      authTag: authTag.toString('hex')
    };
  }

  decrypt(encryptedData) {
    const decipher = crypto.createDecipher(
      this.algorithm,
      this.secretKey
    );
    
    decipher.setAuthTag(Buffer.from(encryptedData.authTag, 'hex'));
    
    let decrypted = decipher.update(encryptedData.encrypted, 'hex', 'utf8');
    decrypted += decipher.final('utf8');
    
    return decrypted;
  }
}

// 使用示例
const encryption = new EncryptionService();

// 加密敏感数据
const encryptedPhone = encryption.encrypt(user.phone);

// 解密敏感数据
const decryptedPhone = encryption.decrypt(encryptedPhone);
```
